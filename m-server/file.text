const express = require('express');
const router = express.Router();
const Aireplay = require('../modules/Chat');  // ✅ Correct model import

// POST /chat - auto-reply from MongoDB
router.post('/', async (req, res) => {
  const { message } = req.body;
  if (!message) return res.status(400).json({ reply: "No message received" });

  try {
    console.log("🔍 Searching for:", message);

    // ✅ Partial + case-insensitive match
    const replyDoc = await Aireplay.findOne({
      userMessage: { $regex: message, $options: 'i' }
    });

    if (replyDoc) {
      console.log("✅ Found:", replyDoc);
      return res.json({ reply: replyDoc.aiReply });
    } else {
      console.log("❌ No match found");
      return res.json({ reply: "🤖 Sorry, I don't have a reply for that yet." });
    }
  } catch (err) {
    console.error("Error finding reply:", err);
    return res.status(500).json({ reply: "⚠️ Something went wrong!" });
  }
});

module.exports = router;

  userMessage: { type: String, required: true },
  aiReply: { type: String, required: true }
});
